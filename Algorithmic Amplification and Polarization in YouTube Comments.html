  return(mean(data[indices] >= 2))

# We create this function to get a bootstrapped CI for a group
get_boot_ci <- function(likes_vector) {
  res <- boot(data = likes_vector, statistic = prop_2plus_func, R = 1000)
  ci <- boot.ci(res, type = "perc")
  return(data.frame(
    observed = res$t0,
    low = ci$percent[4],
    high = ci$percent[5]
  ))
}

# Set seed for reproducibility
set.seed(123)

# Compile the bootstrapped results across all Video Categories
category_results <- yt_data_prop %>%
